# Scripted effects for Venezuela's army reform and doctrine tree

# Effect: VEN_advance_army_reform
# Purpose: Steps Venezuela through the 4-stage outdated army idea chain up to VEN_rejuvinated_army.
VEN_advance_army_reform = {
	if = {
		limit = { has_idea = VEN_outdated_army_idea }
		swap_ideas = {
			remove_idea = VEN_outdated_army_idea
			add_idea = VEN_outdated_army_idea_1
		}
	}
	else_if = {
		limit = { has_idea = VEN_outdated_army_idea_1 }
		swap_ideas = {
			remove_idea = VEN_outdated_army_idea_1
			add_idea = VEN_outdated_army_idea_2
		}
	}
	else_if = {
		limit = { has_idea = VEN_outdated_army_idea_2 }
		swap_ideas = {
			remove_idea = VEN_outdated_army_idea_2
			add_idea = VEN_outdated_army_idea_3
		}
	}
	else_if = {
		limit = { has_idea = VEN_outdated_army_idea_3 }
		swap_ideas = {
			remove_idea = VEN_outdated_army_idea_3
			add_idea = VEN_rejuvinated_army
		}
	}
	else_if = {
		limit = { NOT = { has_idea = VEN_rejuvinated_army } }
		add_ideas = VEN_rejuvinated_army
	}
}

# Effect: VEN_add_land_mastery_20
# Purpose: Grants 20 land doctrine mastery if a grand doctrine is selected, otherwise 35 army experience.
VEN_add_land_mastery_20 = {
	if = {
		limit = { has_selected_land_grand_doctrine = yes }
		add_mastery = {
			amount = 20
			folder = land
		}
	}
	else = {
		army_experience = 35
	}
}
